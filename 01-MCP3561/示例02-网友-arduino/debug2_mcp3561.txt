MCP3561::writeRegisterDefaults( )  

  
  //设置 CONFIG0  ， 使用外部clk， adc为转换模式
  command_byte = CONFIG0_WRITE;  //01 0001 10， 递增写指令，起始地址为config0

  data_byte = CONFIG0_CLK_SEL_EXT;                //bit5-4=00， 外部clk
  data_byte |= CONFIG0_ADC_MODE_CONV;      //bit1-0 = 11， adc模式为转换模式
  SPI.transfer(command_byte);  //0x46， 写config0
  SPI.transfer(data_byte);          //0x03   ，外部clk，转换模式
 

  //设置 CONFIG1  ，
  command_byte = CONFIG1_WRITE;     // 01 0010 10， 递增写指令，起始地址为config1  
  data_byte = CONFIG1_OSR_256;          
  SPI.transfer(command_byte);    //0x4A ， 写config1
  SPI.transfer(data_byte);           //0x0c ， OSR 256
 
  // Next, write to CONFIG2 register. No need to change anything, defaults OK.

  
  command_byte = CONFIG3_WRITE;   // 01 0100 10， 递增写指令，起始地址为config3   
  data_byte = CONFIG3_CONV_MODE_CONTINUOUS; 
  SPI.transfer(command_byte);     //0x52 ， 写config3
  SPI.transfer(data_byte);             //0xc0 ， 连续转换， 24bit，无offcal校验，无gaincal校验
 

  // Next, write to the IRQ register to enable a conversion to trigger an interrupt.
  command_byte = IRQ_WRITE;      // 01 0101 10， 递增写指令，起始地址为 irq  
  data_byte = IRQ_MODE_HIGH;
   SPI.transfer(command_byte);     //0x56 ， 写irq
  SPI.transfer(data_byte);              //0x77 ，
 
}

MCP3564_Read( pBuffer,  Cmd, NumByteToRead)
  
    MCP3564_CS_LOW();

    MCP3564_SendByte(Cmd);    //从寄存器01开始读数据，一共读27个字节的数据

    while (NumByteToRead--)

        *pBuffer = MCP3564_SendByte(Dummy_Byte);   //读27次
        pBuffer++;
    
    MCP3564_CS_HIGH();

MCP3564_Write( pBuffer,  Cmd,  NumByteToWrite)

    MCP3564_CS_LOW();
    
    MCP3564_SendByte(Cmd);

    while (NumByteToWrite--)
    
        MCP3564_SendByte(*pBuffer);
        pBuffer++;
    

    MCP3564_CS_HIGH();

功能：往器件01的芯片，发送cmd为00的无效指令，就会返回status

MCP3564_GetStatus( )

    u8 status=0;
    
    MCP3564_CS_LOW();

    status=MCP3564_SendByte(FASTCMD_UNUSE);    //01 00 00  00 ， 读取器件01的status //主要看后面两位cmd为00

    MCP3564_CS_HIGH();
    
    return status;
